<objective>
Implement OUT-03 + OUT-04: after a cron/heartbeat agent run completes (via
`adapterRegistry.resolve()` in `src/auto-reply/reply/agent-runner-execution.ts`),
evaluate whether the output is worth notifying the user. If not, replace the
reply text with `SILENT_REPLY_TOKEN` so the existing `normalizeReplyPayload()`
pipeline calls `onSkip("silent")` and nothing reaches the channel.

All file paths in this plan are relative to `/home/nikolas/Documents/CODE/AI/minion/`
(the minion gateway subproject). This plan depends on plan 15-01 (`forcedToolCall`)
and composes with plan 15-02 (reuses `evaluateHeartbeatStructured` when
`isHeartbeatStructuredEnabled` is true to avoid double-LLM-calls on heartbeats).

Purpose: Per 15-RESEARCH.md §4, routine cron jobs (status polls that found
nothing, heartbeats with no action) currently send their full response to the
user channel. This spams the user with "nothing to report"-type messages. A
post-run forced-tool evaluation suppresses these silently.

Output: a new `shouldNotifyUser()` evaluator, a suppression bridge, unit tests,
and a single integration-point edit in `agent-runner-execution.ts` that wires
the evaluation into the adapter-resolve path (around line 247).

    ```typescript
    import { Type } from "@sinclair/typebox";
    import type { Context, Model } from "@mariozechner/pi-ai";
    import { forcedToolCall, ForcedToolCallError } from "../../agents/llm-tools/index.js";
    import { SILENT_REPLY_TOKEN } from "../tokens.js";

    export const NOTIFICATION_EVAL_TOOL_NAME = "notification_decision";

    export type NotificationDecision = {
      should_notify: boolean;
      reason: string;
      summary?: string;
    };

    const NotificationDecisionSchema = Type.Object({
      should_notify: Type.Boolean({
        description:
          "true if the output contains something genuinely actionable or informative the user should see; false if the run produced only routine no-op output (no changes, nothing to report, polled-and-found-nothing, etc.)",
      }),
      reason: Type.String({
        description: "Short (<=120 chars) justification. Not shown to user — used for logs.",
      }),
      summary: Type.Optional(
        Type.String({
          description:
            "Optional one-sentence summary to send INSTEAD of the full output when should_notify is true and the full output is verbose. Leave empty to send the full output unchanged.",
        }),
      ),
    });

    export type ShouldNotifyUserOptions = {
      model: Model<"openai-completions">;
      /**
       * The agent's full output text from the just-completed run. This is what
       * the evaluator judges — was it worth notifying the user about?
       */
      output: string;
      /** Optional original user/cron prompt for context — helps the LLM judge relevance. */
      triggerPrompt?: string;
      apiKey?: string;
      timeoutMs?: number;
    };

    /**
     * OUT-03: evaluate whether a routine agent run's output is worth a user
     * notification. Used after cron/heartbeat runs where silence is often the
     * right behavior.
     *
     * FAIL-OPEN: on any evaluator error, defaults to `should_notify: true`.
     * This is the opposite of heartbeat's fail-closed because missing a real
     * notification is a functional regression, whereas missing a heartbeat
     * "I have nothing to report" ack is not.
     */
    export async function shouldNotifyUser(
      opts: ShouldNotifyUserOptions,
    ): Promise<NotificationDecision> {
      // Short-circuit: empty output is handled by normalizeReplyPayload's
      // existing onSkip("empty") path — no need to spend an LLM call.
      if (!opts.output || !opts.output.trim()) {
        return { should_notify: false, reason: "empty output — handled by normalize" };
      }

      const systemPrompt = [
        "You are evaluating whether a routine agent run produced output worth notifying the user about.",
        "Call the notification_decision tool exactly once with your decision.",
        "Default to should_notify:false when the output is a routine status check with no actionable change.",
        "Default to should_notify:true when the output contains new information, a completed task, an alert, or a user-directed question.",
      ].join(" ");

      const userMessage = [
        opts.triggerPrompt ? `Trigger: ${opts.triggerPrompt}\n\n` : "",
        `Agent output:\n${opts.output}`,
      ].join("");

      const context: Omit<Context, "tools"> = {
        messages: [
          { role: "user" as const, content: userMessage, timestamp: Date.now() },
        ],
      };

      try {
        const result = await forcedToolCall<NotificationDecision>({
          model: opts.model,
          context,
          systemPrompt,
          tool: {
            name: NOTIFICATION_EVAL_TOOL_NAME,
            description:
              "Return the notification decision for this agent run. Call exactly once.",
            parameters: NotificationDecisionSchema,
          },
          apiKey: opts.apiKey,
          maxTokens: 200,
          temperature: 0,
          timeoutMs: opts.timeoutMs ?? 20_000,
        });
        return {
          should_notify: Boolean(result.should_notify),
          reason: typeof result.reason === "string" ? result.reason : "",
          summary: typeof result.summary === "string" && result.summary.trim() ? result.summary : undefined,
        };
      } catch (err) {
        const code = err instanceof ForcedToolCallError ? err.code : "UNKNOWN";
        return {
          should_notify: true,
          reason: `notification eval ${code === "TIMEOUT" ? "timeout" : "error"} — fail-open: notifying by default`,
        };
      }
    }

    /**
     * OUT-04: convert a notification decision into the reply text that will
     * flow through normalizeReplyPayload. When should_notify is false, returns
     * SILENT_REPLY_TOKEN so isSilentReplyText() triggers onSkip("silent").
     * When should_notify is true and `summary` is provided, returns the summary
     * instead of the original verbose output. Otherwise returns the original
     * output unchanged.
     */
    export function applyNotificationDecision(
      originalReplyText: string,
      decision: NotificationDecision,
    ): string {
      if (!decision.should_notify) {
        return SILENT_REPLY_TOKEN;
      }
      if (decision.summary && decision.summary.trim()) {
        return decision.summary;
      }
      return originalReplyText;
    }
    ```

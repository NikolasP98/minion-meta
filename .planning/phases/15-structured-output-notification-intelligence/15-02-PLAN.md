<objective>
Implement OUT-02: replace the brittle text-matching heartbeat decision
(`HEARTBEAT_OK` token string) with a forced tool call that returns a typed
`{should_act: boolean, reason: string}` decision. The legacy token path
remains the default; the structured path activates only when
`config.agents.defaults.heartbeat.structured: true` is set.

All file paths in this plan are relative to `/home/nikolas/Documents/CODE/AI/minion/`
(the minion gateway subproject). This plan depends on plan 15-01 (`forcedToolCall`).

Purpose: Per 15-RESEARCH.md §1, text token matching is brittle — the model can
say "I have no HEARTBEAT_OK items but will check again" and the token gets
stripped incorrectly. A forced tool call eliminates that class of failure
because the model MUST return structured data. Keeping `HEARTBEAT_OK` as a
fallback avoids breaking existing deployed agents while we roll out the new
path behind a config flag.

Output: a new `evaluateHeartbeatStructured()` function, a `structured` boolean
field added to `HeartbeatSchema`, an `isHeartbeatStructuredEnabled(config)`
helper, and unit tests. Wiring into the actual heartbeat runner (i.e.,
deciding at runtime which path to take) is the second task in this plan.
</objective>

    import { Type } from "@sinclair/typebox";
    import { forcedToolCall, ForcedToolCallError } from "../agents/llm-tools/index.js";
    import type { Context, Model } from "@mariozechner/pi-ai";

    export const HEARTBEAT_STRUCTURED_TOOL_NAME = "heartbeat_decision";

    export type HeartbeatDecision = {
      should_act: boolean;
      reason: string;
    };

    const HeartbeatDecisionSchema = Type.Object({
      should_act: Type.Boolean({
        description: "true if this heartbeat has genuinely actionable content worth notifying the user; false if nothing needs attention.",
      }),
      reason: Type.String({
        description: "Short (<=120 chars) explanation of why should_act is true or false. Not shown to the user — used for logs and debugging.",
      }),
    });

    export type EvaluateHeartbeatStructuredOptions = {
      model: Model<"openai-completions">;
      /** The assembled heartbeat context (system + user messages). */
      context: Omit<Context, "tools">;
      /** Optional API key override. */
      apiKey?: string;
      /** Defaults to 20_000 — heartbeat decisions should be fast. */
      timeoutMs?: number;
    };

    /**
     * OUT-02: evaluate whether a heartbeat run has produced something worth
     * notifying the user about, via a forced tool call. Fail-closed: any
     * error from the LLM call resolves to `should_act: false` so we default
     * to suppression rather than user spam.
     */
    export async function evaluateHeartbeatStructured(
      opts: EvaluateHeartbeatStructuredOptions,
    ): Promise<HeartbeatDecision> {
      try {
        const result = await forcedToolCall<HeartbeatDecision>({
          model: opts.model,
          context: opts.context,
          tool: {
            name: HEARTBEAT_STRUCTURED_TOOL_NAME,
            description:
              "Decide whether this heartbeat turn requires user notification. Call this tool exactly once with your decision.",
            parameters: HeartbeatDecisionSchema,
          },
          apiKey: opts.apiKey,
          maxTokens: 128,
          temperature: 0,
          timeoutMs: opts.timeoutMs ?? 20_000,
        });
        return {
          should_act: Boolean(result.should_act),
          reason: typeof result.reason === "string" ? result.reason : "",
        };
      } catch (err) {
        const code = err instanceof ForcedToolCallError ? err.code : "UNKNOWN";
        return {
          should_act: false,
          reason: `structured eval ${code === "TIMEOUT" ? "timeout" : "error"}, suppressing by default`,
        };
      }
    }
    ```

    /**
     * OUT-02: Check if structured heartbeat decisions are enabled via
     * `config.agents.defaults.heartbeat.structured: true`. When false/undefined,
     * the legacy HEARTBEAT_OK text-token path remains active.
     *
     * This is a narrow accessor so callers don't need to walk the config tree
     * and we have a single grep-able check for the feature flag.
     */
    export function isHeartbeatStructuredEnabled(config: unknown): boolean {
      if (!config || typeof config !== "object") return false;
      const agents = (config as { agents?: unknown }).agents;
      if (!agents || typeof agents !== "object") return false;
      const defaults = (agents as { defaults?: unknown }).defaults;
      if (!defaults || typeof defaults !== "object") return false;
      const heartbeat = (defaults as { heartbeat?: unknown }).heartbeat;
      if (!heartbeat || typeof heartbeat !== "object") return false;
      return (heartbeat as { structured?: unknown }).structured === true;
    }

    /**
     * OUT-02 suppression bridge: given a heartbeat reply text and a structured
     * decision, return either SILENT_REPLY_TOKEN (so normalizeReplyPayload will
     * call onSkip("silent")) or the original text unchanged.
     *
     * Callers flow the returned string through the normal reply pipeline —
     * suppression happens in normalizeReplyPayload via isSilentReplyText().
     */
    export function applyHeartbeatStructuredDecision(
      replyText: string,
      decision: { should_act: boolean; reason: string },
    ): string {
      if (!decision.should_act) {
        return SILENT_REPLY_TOKEN;
      }
      return replyText;
    }
    ```

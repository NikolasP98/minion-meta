    ```typescript
    import {
      complete,
      type Context,
      type Model,
      type OpenAICompletionsOptions,
      type Tool,
    } from "@mariozechner/pi-ai";
    import type { TSchema } from "@sinclair/typebox";

    export type ForcedToolCallOptions = {
      /** Model instance from getModel(...) — same shape as model-scan.ts uses. */
      model: Model<"openai-completions">;
      /** Context (messages) to send. Tools array is set by this utility. */
      context: Omit<Context, "tools">;
      /** Tool the LLM MUST call. `parameters` is a TypeBox schema. */
      tool: { name: string; description: string; parameters: TSchema };
      /** Optional system prompt prepended to messages. */
      systemPrompt?: string;
      /** API key override. Defaults to complete()'s resolution. */
      apiKey?: string;
      /** Defaults to 256. Structured decisions are small — keep tokens low. */
      maxTokens?: number;
      /** Defaults to 0. Deterministic by default for decision calls. */
      temperature?: number;
      /** Defaults to 30_000. Throws TIMEOUT error if exceeded. */
      timeoutMs?: number;
    };

    export class ForcedToolCallError extends Error {
      readonly code: "NO_TOOL_CALL" | "WRONG_TOOL" | "TIMEOUT" | "PROVIDER_ERROR";
      readonly toolName: string;
      constructor(code: ForcedToolCallError["code"], toolName: string, message: string) {
        super(message);
        this.name = "ForcedToolCallError";
        this.code = code;
        this.toolName = toolName;
      }
    }

    /**
     * Call the LLM with `toolChoice: "required"` forcing it to invoke a specific
     * tool, and return the parsed tool input typed as `T`.
     *
     * Wraps complete() from @mariozechner/pi-ai. Mirrors the pattern from
     * src/agents/models/model-scan.ts:260-281 but generalizes it for typed
     * structured output.
     */
    export async function forcedToolCall&lt;T&gt;(opts: ForcedToolCallOptions): Promise&lt;T&gt; {
      const timeoutMs = opts.timeoutMs ?? 30_000;
      const maxTokens = opts.maxTokens ?? 256;
      const temperature = opts.temperature ?? 0;

      const tool: Tool = {
        name: opts.tool.name,
        description: opts.tool.description,
        parameters: opts.tool.parameters,
      };

      const messages = opts.systemPrompt
        ? [{ role: "system" as const, content: opts.systemPrompt, timestamp: Date.now() }, ...opts.context.messages]
        : opts.context.messages;

      const contextWithTool: Context = {
        messages,
        tools: [tool],
      };

      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(), timeoutMs);
      try {
        const message = await complete(opts.model, contextWithTool, {
          apiKey: opts.apiKey,
          maxTokens,
          temperature,
          toolChoice: "required",
          signal: controller.signal,
        } satisfies OpenAICompletionsOptions);

        const toolCallBlock = message.content.find(
          (block): block is Extract&lt;typeof block, { type: "toolCall" }&gt; => block.type === "toolCall",
        );
        if (!toolCallBlock) {
          throw new ForcedToolCallError(
            "NO_TOOL_CALL",
            opts.tool.name,
            `LLM returned no tool call for forced tool '${opts.tool.name}'`,
          );
        }
        if (toolCallBlock.name !== opts.tool.name) {
          throw new ForcedToolCallError(
            "WRONG_TOOL",
            opts.tool.name,
            `LLM called '${toolCallBlock.name}' instead of required '${opts.tool.name}'`,
          );
        }
        return toolCallBlock.input as T;
      } catch (err) {
        if (err instanceof ForcedToolCallError) throw err;
        if (controller.signal.aborted) {
          throw new ForcedToolCallError(
            "TIMEOUT",
            opts.tool.name,
            `forcedToolCall('${opts.tool.name}') exceeded ${timeoutMs}ms`,
          );
        }
        throw new ForcedToolCallError(
          "PROVIDER_ERROR",
          opts.tool.name,
          err instanceof Error ? err.message : String(err),
        );
      } finally {
        clearTimeout(timer);
      }
    }
    ```
